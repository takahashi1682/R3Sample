問31
public void OnStartButtonClicked()
{
    if (_isRunning) return;

    OnStart.OnNext(Unit.Default);
    _cts = new CancellationTokenSource();
    TimerTask(_cts).Forget();
    _isRunning = true;
}

public void OnCancelButtonClicked()
{
    _currentTime.Value = 0;
    _isRunning = false;
    _cts.Cancel();
}

private async UniTask TimerTask(CancellationTokenSource token)
{
    while (_currentTime.Value < _maxInterval && !token.IsCancellationRequested)
    {
        _currentTime.Value += Time.deltaTime;
        await UniTask.Yield(token.Token);
    }

    OnComplete.OnNext(Unit.Default);
}

private async UniTask SpawnBlocks()
{
    var offsetX = (_width - 1) / 2f;

    for (var y = 0; y < _height; y++)
    {
        for (var x = 0; x < _width; x++)
        {
            var pos = new Vector3(x - offsetX, _spawnHeight + y, 0);
            Instantiate(_blockPrefab, pos, Quaternion.identity);
            await UniTask.Delay(TimeSpan.FromSeconds(_spawnInterval));
        }
    }
